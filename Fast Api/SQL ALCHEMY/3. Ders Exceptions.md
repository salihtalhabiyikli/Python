@app.post("/kayit")
def kayitOl(kullanici: kullaniciOlustur,db: Session= Depends(db_al)):

	mevcut=db.query(Kullanici).filter(Kullanici.email==kullanici.email).first()
	# Bu kısım eşleşeni bulup ilk sonucu getiriyor 
		
	if mevcut=True:
		raise HTTPException(
		status_code=400,
		detail="Bu mail zaten kayıtlı"
		)
		
	yeni_kullanici=Kullanci(
	ad= kullanici.ad
	email=kullanici.email,
	sifre=kullanici.sifre
	)
	
	db.add(yeni_kullanici)
	db.commit()
	db.refresh(yeni_kullanici)
	
return yeni_kullanici
