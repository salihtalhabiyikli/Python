### Update
```python
@app.put("/todolar/{id}", response_model=TodoResponse)
def todo_guncelle(id: int, guncelleme: TodoGuncelle, db: Session = Depends(db_al)):

    # Todo var mı?
    todo = db.query(Todo).filter(Todo.id == id).first()
    if not todo:
        raise HTTPException(status_code=404, detail="Todo bulunamadı")

    # Sadece gönderilen alanları güncelle
    for alan, deger in guncelleme.model_dump(exclude_unset=True).items():
        setattr(todo, alan, deger)

    db.commit()
    db.refresh(todo)
    return todo
```

### Delete 
```python
@app.delete("/todolar/{id}")
def todo_sil(id: int, db: Session = Depends(db_al)):

    todo = db.query(Todo).filter(Todo.id == id).first()
    if not todo:
        raise HTTPException(status_code=404, detail="Todo bulunamadı")

    db.delete(todo)
    db.commit()
    return {"mesaj": "Todo silindi"}
```
